  330:   const lines: string[] = [];

  690: function installRoute(
  691:   text: string,
  692:   installedUrl: string,
  693:   replaceExistingRoute: boolean,
  694: ): { text: string; previous: CodexIntegrationJournal["previous"] } {
  695:   const document = parseDocument(text);
  696:   const previous = assignments(document.lines);
  697:   const conflicts = (Object.entries(previous) as Array<[ManagedAssignmentKey, PreviousAssignment]>)
  698:     .filter(([key, assignment]) => assignment.present
  699:       && !(key === "model_provider" && assignment.value === "openai"))
  700:     .map(([key, assignment]) => `${key}=${JSON.stringify(assignment.value)}`);
  701:   if (conflicts.length > 0 && !replaceExistingRoute) {
  702:     throw new Error(
  703:       `Codex already configures model routing (${conflicts.join(", ")}). `
  704:       + "Rerun with --replace-codex-route to replace it reversibly.",
  705:     );
  706:   }
  707: 
  708:   for (const key of ["model_provider", "model_catalog_json"] as const) {
  709:     const location = findTopLevelAssignment(document.lines, key);
  710:     if (location.index !== undefined) removeDocumentLine(document, location.index);
  711:   }
  712:   const currentBaseUrl = findTopLevelAssignment(document.lines, "openai_base_url");
  713:   if (currentBaseUrl.index !== undefined) {
  714:     document.lines[currentBaseUrl.index] = `openai_base_url = ${JSON.stringify(installedUrl)}`;
  715:   } else {
  716:     insertDocumentLine(document, firstTableIndex(document.lines), `openai_base_url = ${JSON.stringify(installedUrl)}`);
  717:   }
  718:   removeManagedComment(document);
  719:   const installedBaseUrl = findTopLevelAssignment(document.lines, "openai_base_url");
  720:   insertDocumentLine(document, installedBaseUrl.index!, MANAGED_COMMENT);
  721:   return { text: renderDocument(document), previous };
  722: }

  724: function verifyInstalledRoute(text: string, journal: ManagedRouteJournal): void {
  725:   const lines = splitLines(text);
  726:   const current = assignments(lines);
  727:   if (current.openai_base_url.value !== journal.installed.openai_base_url) {
  728:     throw new Error("Codex openai_base_url changed after setup; refusing to overwrite the user's newer value");
  729:   }
  730:   if (current.model_provider.present || current.model_catalog_json.present) {
  731:     throw new Error("Codex model_provider or model_catalog_json changed after setup; refusing to overwrite the user's newer value");
  732:   }
  733:   if (!lines.includes(MANAGED_COMMENT)) {
  734:     throw new Error("Managed Codex route marker changed after setup; refusing to overwrite it");
  735:   }
  736:   if (journal.version === 5 || journal.version === 6) verifyInstalledFeatures(text, journal);
  737: }

  739: function previousAssignmentMatches(current: PreviousAssignment, previous: PreviousAssignment): boolean {
  740:   return current.present === previous.present
  741:     && (!current.present || current.value === previous.value);

  744: function verifyRestoredRoute(
  745:   text: string,
  746:   journal: CodexIntegrationJournal | LegacyCodexIntegrationJournalV5 | LegacyCodexIntegrationJournalV4,
  747: ): void {
  748:   const lines = splitLines(text);
  749:   const current = assignments(lines);
  750:   for (const key of ["openai_base_url", "model_provider", "model_catalog_json"] as const) {
  751:     if (!previousAssignmentMatches(current[key], journal.previous[key])) {
  752:       throw new Error(`Codex ${key} changed while the bridge was disconnected; refusing to overwrite the user's newer value`);
  753:     }
  754:   }
  755:   if (lines.includes(MANAGED_COMMENT)) {
  756:     throw new Error("Managed Codex route marker is present while the bridge is disconnected");
  757:   }
  758:   if (journal.version === 5 || journal.version === 6) {
  759:     const previousFeatures: Array<readonly [string, PreviousFeatureAssignment]> = [
  760:       ["remote_compaction_v2", journal.previousRemoteCompactionV2],
  761:       ["multi_agent", journal.previousMultiAgent],
  762:     ];
  763:     if (journal.version === 6) {
  764:       previousFeatures.push(["multi_agent_v2", journal.previousMultiAgentV2]);
  765:     }
  766:     for (const [key, previous] of previousFeatures) {
  767:       const current = key === "multi_agent_v2"
  768:         ? findMultiAgentV2Assignment(lines)
  769:         : findFeatureAssignment(lines, key);
  770:       const matches = current.present === previous.present
  771:         && (current.tableName ?? "features") === (previous.tableName ?? "features")
  772:         && (!current.present || current.rawLine === previous.rawLine);
  773:       if (!matches) {
  774:         throw new Error(
  775:           `Codex [features].${key} changed while the bridge was disconnected; refusing to overwrite the user's newer value`,
  776:         );
  777:       }
  778:     }
  779:   }
  780: }

  817: function restoreManagedRoute(text: string, journal: ManagedRouteJournal): string {
  818:   verifyInstalledRoute(text, journal);
  819:   const document = parseDocument(text);
  820:   removeManagedComment(document);
  821:   const currentBaseUrl = findTopLevelAssignment(document.lines, "openai_base_url");
  822:   if (currentBaseUrl.index === undefined) throw new Error("Managed Codex openai_base_url is missing");
  823:   const previousBaseUrl = journal.previous.openai_base_url;
  824:   if (previousBaseUrl.present) {
  825:     if (!previousBaseUrl.rawLine) throw new Error("Codex integration journal is missing the prior openai_base_url line");
  826:     document.lines[currentBaseUrl.index] = previousBaseUrl.rawLine;
  827:   } else {
  828:     removeDocumentLine(document, currentBaseUrl.index);
  829:   }
  830:   const removedAssignments = (["model_provider", "model_catalog_json"] as const)
  831:     .map(key => ({ key, previous: journal.previous[key] }))
  832:     .filter(item => item.previous.present)
  833:     .sort((left, right) => (left.previous.index ?? Number.MAX_SAFE_INTEGER) - (right.previous.index ?? Number.MAX_SAFE_INTEGER));
  834:   for (const item of removedAssignments) {
  835:     if (!item.previous.rawLine) throw new Error(`Codex integration journal is missing the prior ${item.key} line`);
  836:     const index = Math.min(item.previous.index ?? firstTableIndex(document.lines), firstTableIndex(document.lines));
  837:     insertDocumentLine(document, index, item.previous.rawLine);
  838:   }
  839:   const restoredRoute = renderDocument(document);
  840:   return journal.version === 5 || journal.version === 6
  841:     ? restoreManagedFeatures(restoredRoute, journal)
  842:     : restoredRoute;
  843: }

  845: function restoreLegacyV2(text: string, journal: LegacyCodexIntegrationJournal): string {
  846:   if (!text.includes(journal.providerBlock)) {
  847:     throw new Error("Managed legacy Codex provider block changed after setup; refusing migration");
  848:   }
  849:   const withoutProvider = text.replace(journal.providerBlock, "").replace(/\n{3,}/g, "\n\n");
  850:   const document = parseDocument(withoutProvider);
  851:   for (const key of ["model_provider", "model_catalog_json"] as const) {
  852:     const current = findTopLevelAssignment(document.lines, key);
  853:     if (current.value !== journal.installed[key] || current.index === undefined) {
  854:       throw new Error(`Managed legacy Codex ${key} changed after setup; refusing migration`);
  855:     }
  856:     const previous = journal.previous[key];
  857:     if (previous.present) {
  858:       if (!previous.rawLine) throw new Error(`Legacy Codex integration journal is missing the prior ${key} line`);
  859:       document.lines[current.index] = previous.rawLine;
  860:     } else {
  861:       removeDocumentLine(document, current.index);
  862:     }
  863:   }
  864:   removeManagedComment(document);
  865:   const restoredCatalog = findTopLevelAssignment(document.lines, "model_catalog_json");
  866:   if (restoredCatalog.index !== undefined && restoredCatalog.value && !existsSync(resolve(restoredCatalog.value))) {
  867:     removeDocumentLine(document, restoredCatalog.index);
  868:   }
  869:   return renderDocument(document);
  870: }

  872: function readJournal(): AnyCodexIntegrationJournal | undefined {
  873:   const path = getCodexJournalPath();
  874:   if (!existsSync(path)) return undefined;
  875:   const value = JSON.parse(readFileSync(path, "utf8")) as Record<string, unknown>;
  876:   if (value.version === 6
  877:     && typeof value.active === "boolean"
  878:     && value.installed
  879:     && value.previous
  880:     && value.previousRemoteCompactionV2
  881:     && value.previousMultiAgent
  882:     && value.previousMultiAgentV2
  883:     && typeof value.configPath === "string") {
  884:     return value as unknown as CodexIntegrationJournal;
  885:   }
  886:   if (value.version === 5
  887:     && typeof value.active === "boolean"
  888:     && value.installed
  889:     && value.previous
  890:     && value.previousRemoteCompactionV2
  891:     && value.previousMultiAgent
  892:     && typeof value.configPath === "string") {
  893:     return value as unknown as LegacyCodexIntegrationJournalV5;
  894:   }
  895:   if (value.version === 4
  896:     && typeof value.active === "boolean"
  897:     && value.installed
  898:     && value.previous
  899:     && typeof value.configPath === "string") {
  900:     return value as unknown as LegacyCodexIntegrationJournalV4;
  901:   }
  902:   if (value.version === 3 && value.installed && value.previous && typeof value.configPath === "string") {
  903:     return value as unknown as LegacyCodexIntegrationJournalV3;
  904:   }
  905:   if (value.version === 2 && value.installed && value.previous && typeof value.providerBlock === "string") {
  906:     return value as unknown as LegacyCodexIntegrationJournal;
  907:   }
  908:   throw new Error(`Invalid Codex integration journal: ${path}`);
  909: }

  926: export function preflightCodexIntegration(
  927:   config: AppConfig,
  928:   options: InstallCodexIntegrationOptions = {},
  929: ): void {
  930:   const configPath = getCodexConfigPath();
  931:   const currentText = existsSync(configPath) ? readFileSync(configPath, "utf8") : "";
  932:   const existing = readJournal();
  933:   const installedUrl = routeUrl(config);
  934:   if (existing) assertJournalTargetsConfig(existing, configPath);
  935:   if (existing?.version === 3 || existing?.version === 4 || existing?.version === 5 || existing?.version === 6) {
  936:     if ((existing.version === 4 || existing.version === 5 || existing.version === 6) && !existing.active) {
  937:       verifyRestoredRoute(currentText, existing);
  938:     }
  939:     else verifyInstalledRoute(currentText, existing);
  940:     return;
  941:   }
  942:   let baseline = currentText;
  943:   if (existing?.version === 2) {
  944:     if (existsSync(existing.catalogPath) && sha256(readFileSync(existing.catalogPath)) !== existing.catalogSha256) {
  945:       throw new Error(`Managed legacy catalog changed after setup; refusing migration: ${existing.catalogPath}`);
  946:     }
  947:     baseline = restoreLegacyV2(currentText, existing);
  948:   }
  949:   installIntegrationConfig(baseline, installedUrl, options.replaceExistingRoute === true);
  950: }

 1010:       { path: configPath, data: installedText },

 1018:     let previousRemoteCompactionV2 = existing.previousRemoteCompactionV2;
 1019:     let previousMultiAgent = existing.previousMultiAgent;
 1020:     let previousMultiAgentV2: PreviousFeatureAssignment;
 1021:     if (!existing.active) {
 1022:       verifyRestoredRoute(currentText, existing);
 1023:       const patched = installIntegrationConfig(currentText, installedUrl, true);
 1024:       assertPreservedPreviousAssignments(patched.previous, existing.previous);
 1025:       assertPreservedPreviousFeature(
 1026:         patched.previousRemoteCompactionV2,
 1027:         existing.previousRemoteCompactionV2,
 1028:         "remote_compaction_v2",
 1029:       );

 1031:         patched.previousMultiAgent,
 1032:         existing.previousMultiAgent,
 1033:         "multi_agent",
 1034:       );

 1038:       previousMultiAgentV2 = patched.previousMultiAgentV2;
 1039:     } else {
 1040:       const routedText = updateManagedRouteUrl(currentText, existing, installedUrl);
 1041:       const multiAgentV2 = installMultiAgentV2Feature(routedText);
 1042:       installedText = multiAgentV2.text;
 1043:       previousMultiAgentV2 = multiAgentV2.previous;

 1062:       { path: configPath, data: installedText },
 1063:       { path: getCodexJournalPath(), data: `${JSON.stringify(updated, null, 2)}\n` },
 1064:     ], [getCodexModelsCachePath()]);

 1066:   }
 1067: 
 1068:   if (existing?.version === 3 || existing?.version === 4) {
 1069:     let routedText: string;

 1071:       verifyRestoredRoute(currentText, existing);
 1072:       const patched = installRoute(currentText, installedUrl, true);
 1073:       assertPreservedPreviousAssignments(patched.previous, existing.previous);
 1074:       routedText = patched.text;
 1075:     } else {
 1076:       routedText = updateManagedRouteUrl(currentText, existing, installedUrl);

 1079:     const updated: CodexIntegrationJournal = {
 1080:       version: 6,
 1081:       active: true,

  182:     if (!Number.isInteger(options.port) || options.port < 1 || options.port > 65_535) throw new Error("--port must be an integer from 1 to 65535");
  183:     config.port = options.port;
  184:   }
  185:   if (options.chromeExecutablePath) config.chromeExecutablePath = options.chromeExecutablePath;
  186:   if (options.browserHostDescriptorPath) {
  187:     config.browserHost = "launcher";
  188:     config.browserHostDescriptorPath = options.browserHostDescriptorPath;
  189:     config.brokerSocketPath = defaultBrokerEndpoint();
  190:   } else if (options.chromeExecutablePath) {
  191:     config.browserHost = "managed-chrome";
  192:     delete config.browserHostDescriptorPath;
  193:   }
  194:   if (options.appName) config.appName = options.appName;
  195:   if (options.autoApproveToolCalls !== undefined) config.autoApproveToolCalls = options.autoApproveToolCalls;
  196:   if (options.acknowledgedUnofficial) config.acknowledgedUnofficialAt = new Date().toISOString();
  197:   if (!config.acknowledgedUnofficialAt) {
  198:     throw new Error("Setup requires explicit acknowledgement that this is unofficial browser automation. Pass --acknowledge-unofficial.");
  199:   }
  200:   return config;
  201: }

  203: async function configureTunnel(config: AppConfig, existing: AppConfig | undefined, options: SetupOptions): Promise<void> {
  204:   if (config.mode === "browser-only") {
  205:     delete config.tunnel;
  206:     return;
  207:   }
  208:   const existingTunnel = existing?.mode === "full" ? existing.tunnel : undefined;
  209:   const tunnelId = options.tunnelId ?? existingTunnel?.tunnelId;
  210:   if (!tunnelId) {
  211:     throw new Error("Full mode requires --tunnel-id. Create it at https://platform.openai.com/settings/organization/tunnels");
  212:   }
  213:   let runtimeKeyFile = existingTunnel?.runtimeKeyFile;
  214:   if (!runtimeKeyFile && existsSync(managedRuntimeKeyPath())) runtimeKeyFile = managedRuntimeKeyPath();
  215:   if (options.runtimeKeyFile) runtimeKeyFile = installRuntimeKey(options.runtimeKeyFile);
  216:   if (options.runtimeKeyValue) runtimeKeyFile = installRuntimeKeyBytes(options.runtimeKeyValue);
  217:   if (!runtimeKeyFile || !existsSync(runtimeKeyFile)) {
  218:     throw new Error("Full mode requires a runtime key. Import it interactively or pass --runtime-key-file; create it at https://platform.openai.com/settings/organization/api-keys");
  219:   }
  220:   const installedBinary = await installTunnelClient();
  221:   config.tunnel = createTunnelConfig({
  222:     binaryPath: installedBinary,
  223:     tunnelId,
  224:     runtimeKeyFile,
  225:     profileName: existingTunnel?.profileName,
  226:     alias: existingTunnel?.alias,
  227:   });
  228: }

  230: async function bootstrapTunnelProfile(config: AppConfig): Promise<void> {
  231:   let bootstrapError: unknown;
  232:   try {
  233:     // `runtimes connect` writes the native profile and returns success only after its managed
  234:     // runtime is running, healthy, and ready. Setup stops that validation runtime transactionally;
  235:     // the launcher supervisor reconnects the same alias after the new configuration is committed.
  236:     connectTunnel(config);
  237:   } catch (error) {
  238:     bootstrapError = error;
  239:   }
  240:   try {
  241:     stopTunnel(config);
  242:   } catch (stopError) {
  243:     if (bootstrapError) {
  244:       const primary = bootstrapError instanceof Error ? bootstrapError.message : String(bootstrapError);
  245:       const cleanup = stopError instanceof Error ? stopError.message : String(stopError);
  246:       throw new Error(`${primary}; temporary tunnel cleanup also failed: ${cleanup}`);
  247:     }
  248:     throw stopError;
  249:   }
  250:   if (bootstrapError) throw bootstrapError;
  251: }

  253: export async function setup(options: SetupOptions): Promise<SetupResult> {
  254:   const existing = loadExistingConfig();
  255:   const config = baseConfig(existing, options);
  256:   const launcherOwned = config.browserHost === "launcher";
  257:   if (!launcherOwned && process.platform !== "darwin") {
  258:     throw new Error(
  259:       "Terminal-only managed Chrome setup currently requires macOS. "
  260:       + "Use the Codex Web GPT launcher on Windows or Linux.",
  261:     );
  262:   }
  263:   preflightCodexIntegration(config, {
  264:     replaceExistingRoute: options.replaceCodexRoute,
  265:   });
  266:   const refreshTunnelWorker = tunnelWorkerRuntimeChanged(existing, config);
  267:   if (existing && options.restartService) config.controlToken = randomBytes(32).toString("base64url");
  268:   const beforeService = getServiceStatus();
  269:   if (launcherOwned && (beforeService.installed || beforeService.loaded)) {
  270:     if (!existing) {
  271:       throw new Error("A legacy background service exists without a verifiable configuration; refusing automatic migration");
  272:     }
  273:     if (!options.restartService) {
  274:       throw new Error(
  275:         "Launcher ownership migration must stop the legacy background service. "
  276:         + "Retry from the launcher after the active Codex task finishes.",
  277:       );
  278:     }
  279:   }
  280:   if (beforeService.loaded && !existing) {
  281:     throw new Error("A codex-chatgpt-web service is loaded but its configuration is missing; refusing to replace an unverifiable process");
  282:   }
  283: 
  284:   let loginCreated = false;
  285:   let proAvailable: boolean | undefined;
  286:   if (config.browserHost === "launcher") {
  287:     if (options.forceLogin) throw new Error("Launcher browser login is owned by the launcher UI; --login cannot replace it");
  288:     const detectPro = launcherCapabilityProbeRequired(existing);
  289:     const inspected = await inspectLauncherBrowserHost(config.browserHostDescriptorPath!, {
  290:       detectPro,
  291:     });
  292:     proAvailable = detectPro ? inspected.proAvailable : existing!.proAvailable;
  293:   } else {
  294:     proAvailable = storedBrowserLoginCapabilities(config).proAvailable;
  295:     const loginRequired = options.forceLogin || !browserLoginStateExists(config);
  296:     const capabilityProbeRequired = !loginRequired && proAvailable === undefined;
  297:     if (beforeService.loaded && (loginRequired || capabilityProbeRequired) && !options.restartService) {
  298:       throw new Error(
  299:         "Setup must verify the browser account before changing the running daemon. "
  300:         + "Rerun from a normal terminal with --restart-service after the active task finishes.",
  301:       );
  302:     }
  303:     if (beforeService.loaded && (loginRequired || capabilityProbeRequired) && existing) await assertServiceIdle(existing);
  304:     if (loginRequired) {
  305:       const login = await loginToChatGpt(config);
  306:       proAvailable = login.proAvailable;
  307:       loginCreated = true;
  308:     } else if (capabilityProbeRequired) {
  309:       proAvailable = (await inspectBrowserLoginCapabilities(config)).proAvailable;
  310:     }
  311:   }
  312:   config.proAvailable = proAvailable === true;
  313:   const explicitTunnelChange = Boolean(options.tunnelId || options.runtimeKeyFile || options.runtimeKeyValue);
  314:   const preliminaryChange = Boolean(existing && (meaningfulRuntimeChange(existing, config) || explicitTunnelChange || options.forceLogin));
  315:   if (beforeService.loaded && preliminaryChange && !options.restartService) {
  316:     throw new Error(
  317:       "The daemon is currently serving a Codex task and setup would change its runtime. "
  318:       + "Rerun from a normal terminal with --restart-service after the active task finishes.",
  319:     );
  320:   }
  321:   if (beforeService.loaded && preliminaryChange && existing) await assertServiceIdle(existing);
  322:   await configureTunnel(config, existing, options);
  323: 
  324:   const changedWhileLoaded = Boolean(existing && beforeService.loaded && meaningfulRuntimeChange(existing, config));
  325:   if (changedWhileLoaded && !options.restartService) {
  326:     throw new Error(
  327:       "The daemon is currently serving a Codex task and setup would change its runtime. "
  328:       + "Rerun from a normal terminal with --restart-service after the active task finishes.",
  329:     );
  330:   }
  331:   if (changedWhileLoaded && !preliminaryChange && existing) await assertServiceIdle(existing);
  332:   if (!beforeService.loaded) await assertPortAvailable(config.host, config.port);
  333: 
  334:   if (!launcherOwned) {
  335:     saveConfig(config);
  336:     installService(config);
  337:     if (changedWhileLoaded && options.restartService && existing) await restartService(existing);
  338:     await waitForProxy(config);
  339:   }
  340: 
  341:   let tunnelReady: boolean | null = null;
  342:   if (config.mode === "browser-only" && existing?.mode === "full") {
  343:     const previousTunnelService = getTunnelServiceStatus();
  344:     if (previousTunnelService.installed || previousTunnelService.loaded) await uninstallTunnelService();
  345:     stopTunnel(existing);
  346:   }
  347:   if (config.mode === "full") {
  348:     const profilePath = join(config.tunnel!.profileDir, `${config.tunnel!.profileName}.yaml`);
  349:     const tunnelService = getTunnelServiceStatus();
  350:     const needsProfile = !existsSync(profilePath);
  351:     if (launcherOwned) {
  352:       if (tunnelService.installed || tunnelService.loaded) await uninstallTunnelService();
  353:       if (needsProfile || refreshTunnelWorker || explicitTunnelChange) {
  354:         await bootstrapTunnelProfile(config);
  355:       }
  356:     } else {
  357:       const needsOwnershipMigration = !tunnelService.installed || !tunnelService.loaded || !tunnelServiceDefinitionMatches(config);
  358:       if (needsOwnershipMigration || needsProfile) {
  359:         await assertServiceIdle(config);
  360:         if (tunnelService.loaded) await stopTunnelService();
  361:         await bootstrapTunnelProfile(config);
  362:         installTunnelService(config);
  363:       } else if (refreshTunnelWorker) {
  364:         await assertServiceIdle(config);
  365:         await restartTunnelService();
  366:       }
  367:       const status = await waitForTunnelReady(config);
  368:       if (!status.ok) throw new Error(`Tunnel runtime did not become healthy and ready: ${status.detail}`);
  369:       tunnelReady = true;
  370:     }
  371:   }
  372:   if (launcherOwned && (beforeService.installed || beforeService.loaded)) {
  373:     await uninstallService(existing!);
  374:   }
  375:   if (launcherOwned) saveConfig(config);
  376:   // Keep the previous terminal runtime intact through the ownership handoff. A later launcher
  377:   // setup removes it once the launcher-owned configuration is already the established baseline.
  378:   const migratingTerminalRuntime = Boolean(
  379:     launcherOwned && existing && existing.browserHost !== "launcher",
  380:   );
  381:   if (!migratingTerminalRuntime) removeLegacyRuntimeArtifacts(config);
  382:   installCodexIntegration(config, {
  383:     replaceExistingRoute: options.replaceCodexRoute,
  384:   });
  385: 
  386:   return {
  387:     mode: config.mode,
  388:     configPath: getConfigPath(),
  389:     loginCreated,
  390:     serviceLoaded: launcherOwned ? false : getServiceStatus().loaded,
  391:     tunnelReady,
  392:     codexRestartRequired: true,
  393:     connectorSetupRequired: config.mode === "full",
  394:   };
  395: }
